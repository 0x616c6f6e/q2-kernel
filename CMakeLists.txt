cmake_minimum_required(VERSION 3.22)
project(Q2_Kernel VERSION 0.1.0 LANGUAGES C)
if(CMAKE_C_COMPILER MATCHES "zig(\\.exe)?$")
  # Zig exposes archive tools as subcommands rather than standalone executables.
  set(CMAKE_AR "${CMAKE_C_COMPILER}")
  set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> ar qc <TARGET> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> ar q <TARGET> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_FINISH "<CMAKE_AR> ranlib <TARGET>")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(Q2_WITH_STORAGE "LittleFS and FatFs components" ON)
option(Q2_WITH_SQLITE "Memory-only SQLite component" OFF)
option(Q2_WITH_USB "TinyUSB CDC component; consumer supplies DCD" OFF)
option(Q2_WITH_LVGL "Portable LVGL graphics provider" OFF)
option(Q2_ENABLE_CJK_FONT "Embed the Chinese font and enable Tiny TTF" ON)
option(Q2_BUILD_LVGL_TESTS "Test LVGL rendering, input and Lua integration" OFF)
option(Q2_BUILD_TESTS "Build portable host integration tests" OFF)
option(Q2_ENABLE_LOG "Compile Q2 diagnostic logging" ON)
option(Q2_BUILD_APPS_TESTS "Build foreground application manager tests" OFF)
option(Q2_BUILD_LOG_TESTS "Build logging contract tests (also valid with logging OFF)" OFF)
include(cmake/sources.cmake)
add_library(q2_kernel STATIC ${Q2_CORE_SOURCES})
target_include_directories(q2_kernel PUBLIC include PRIVATE src/core)
target_compile_definitions(q2_kernel PUBLIC Q2_ENABLE_LOG=$<BOOL:${Q2_ENABLE_LOG}>)
if(Q2_BUILD_TESTS OR Q2_BUILD_LOG_TESTS)
  enable_testing()
  add_executable(q2_log_tests tests/test_log.c)
  target_link_libraries(q2_log_tests PRIVATE q2_runtime)
  add_test(NAME q2_log_contracts COMMAND q2_log_tests)
endif()
add_library(q2_lua_vendor STATIC ${Q2_LUA_SOURCES} third_party/cjson/cJSON.c)
target_include_directories(q2_lua_vendor PUBLIC third_party/lua third_party/cjson)
target_compile_definitions(q2_lua_vendor PRIVATE LUAI_MAXCCALLS=32 LUAI_MAXSTACK=2048 CJSON_NESTING_LIMIT=16)
target_link_libraries(q2_kernel PRIVATE q2_lua_vendor)
add_library(q2_runtime STATIC ${Q2_RUNTIME_SOURCES})
target_link_libraries(q2_runtime PUBLIC q2_kernel PRIVATE q2_lua_vendor)
if(UNIX)
  target_link_libraries(q2_lua_vendor PUBLIC m)
endif()
add_library(q2_display STATIC components/display/display.c)
target_link_libraries(q2_display PUBLIC q2_kernel)
if(Q2_WITH_LVGL)
  include(cmake/lvgl.cmake)
  set(Q2_CJK_SOURCE "")
  if(Q2_ENABLE_CJK_FONT)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(Q2_CJK_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/chinese_font.c")
    add_custom_command(OUTPUT "${Q2_CJK_SOURCE}"
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_chinese_font.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/q2-cjk-regular.ttf" "${Q2_CJK_SOURCE}"
      DEPENDS scripts/embed_chinese_font.py assets/fonts/q2-cjk-regular.ttf assets/fonts/manifest.json
      VERBATIM)
  endif()
  add_library(q2_lvgl STATIC components/display/lvgl.c components/display/image_decode.c "${Q2_CJK_SOURCE}")
  target_link_libraries(q2_lvgl PUBLIC q2_kernel PRIVATE q2_lvgl_vendor)
endif()
if(Q2_BUILD_LVGL_TESTS)
  if(NOT Q2_WITH_LVGL)
    message(FATAL_ERROR "LVGL tests require Q2_WITH_LVGL=ON")
  endif()
  enable_testing()
  add_executable(q2_lvgl_tests tests/test_lvgl.c)
  target_link_libraries(q2_lvgl_tests PRIVATE q2_runtime q2_lvgl q2_lvgl_vendor)
  add_test(NAME q2_lvgl_contracts COMMAND q2_lvgl_tests)
  add_executable(q2_system_ui_tests tests/test_system_ui.c)
  target_link_libraries(q2_system_ui_tests PRIVATE q2_runtime q2_lvgl)
  add_test(NAME q2_system_ui_contracts COMMAND q2_system_ui_tests)
endif()
if(Q2_WITH_STORAGE)
  add_library(q2_storage STATIC ${Q2_STORAGE_SOURCES})
  target_include_directories(q2_storage PRIVATE third_party/littlefs third_party/fatfs components/storage)
  target_compile_definitions(q2_storage PRIVATE LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
  target_link_libraries(q2_storage PUBLIC q2_kernel)
endif()
if(Q2_WITH_SQLITE)
  add_library(q2_sqlite STATIC ${Q2_SQLITE_SOURCES})
  target_include_directories(q2_sqlite PRIVATE third_party/sqlite)
  target_compile_definitions(q2_sqlite PRIVATE ${Q2_SQLITE_DEFINITIONS})
  target_link_libraries(q2_sqlite PUBLIC q2_kernel)
endif()
if(Q2_WITH_USB)
  add_library(q2_usb STATIC ${Q2_USB_SOURCES})
  target_include_directories(q2_usb PUBLIC components/usb third_party/tinyusb/src)
  target_link_libraries(q2_usb PUBLIC q2_kernel)
endif()
if(Q2_BUILD_TESTS)
  if(NOT Q2_WITH_STORAGE OR NOT Q2_WITH_SQLITE OR NOT Q2_WITH_USB)
    message(FATAL_ERROR "Integration tests require all optional components")
  endif()
  enable_testing()
  add_executable(q2_tests tests/test_kernel.c tests/fake_dcd.c)
  target_link_libraries(q2_tests PRIVATE q2_runtime q2_storage q2_sqlite q2_usb q2_display)
  add_test(NAME q2_kernel_contracts COMMAND q2_tests)
endif()

if(Q2_BUILD_APPS_TESTS)
  enable_testing()
  add_executable(q2_apps_tests tests/test_apps.c)
  target_link_libraries(q2_apps_tests PRIVATE q2_runtime)
  add_test(NAME q2_apps_contracts COMMAND q2_apps_tests)
  add_executable(q2_multi_runtime_tests tests/test_multi_runtime.c)
  target_link_libraries(q2_multi_runtime_tests PRIVATE q2_runtime)
  add_test(NAME q2_multi_runtime_contracts COMMAND q2_multi_runtime_tests)
endif()

  add_executable(q2_keys_tests tests/test_keys.c)
  target_link_libraries(q2_keys_tests PRIVATE q2_runtime)
  add_test(NAME q2_keys_contracts COMMAND q2_keys_tests)

if(Q2_BUILD_APPS_TESTS OR Q2_BUILD_TESTS)

  add_test(NAME q2_data_contracts COMMAND q2_data_tests)

  add_executable(q2_network_tests tests/test_network.c)
  target_link_libraries(q2_network_tests PRIVATE q2_kernel)
  add_test(NAME q2_network_contracts COMMAND q2_network_tests)

if(Q2_BUILD_AUDIO_TESTS)
  enable_testing()
  add_executable(q2_audio_tests tests/test_audio.c)
  target_link_libraries(q2_audio_tests PRIVATE q2_kernel)

if(Q2_BUILD_APPS_TESTS OR Q2_BUILD_TESTS)
  add_executable(q2_lua_budget_tests tests/test_lua_budget.c)
  target_link_libraries(q2_lua_budget_tests PRIVATE q2_runtime)
  foreach(case bootstrap native wall instructions calls mixed)
    add_test(NAME q2_lua_budget_${case} COMMAND q2_lua_budget_tests ${case})
    set_tests_properties(q2_lua_budget_${case} PROPERTIES TIMEOUT 5)
  endforeach()
